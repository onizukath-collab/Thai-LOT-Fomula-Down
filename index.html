function renderPresetPairCollisionSummary() { const meta = document.getElementById('presetPairSummaryMeta'); const content = document.getElementById('presetPairSummaryContent'); if (!meta || !content) return; // 💡 ดึงแฟ้มทั้งหมดและรวมสูตรจากทุกแฟ้มเข้ามาด้วยกันทันที const allFolders = getStoredFolders(); let presets = {}; Object.values(allFolders).forEach(folderPresets => { if (folderPresets && typeof folderPresets === 'object') { presets = { ...presets, ...folderPresets }; } }); // ดึงฟอร์มปัจจุบันบนหน้าจอมาใส่ชั่วคราวเพื่อให้รีเฟรชแล้วเห็นผลทันที const activePresetName = document.getElementById('presetSelect').value; if (activePresetName && presets[activePresetName]) { presets[activePresetName] = getCurrentFormState(); } const names = Object.keys(presets); if (!names.length) { meta.textContent = 'ยังไม่มีสูตรที่บันทึกไว้'; content.innerHTML = '

บันทึกสูตรอย่างน้อย 1 สูตร แล้วจึงจะแสดงผลเปรียบเทียบ

'; return; } const counts = new Map(); names.forEach(name => { const pairsInThisPreset = getPassedPairsFromPresetState(presets[name]); pairsInThisPreset.forEach(pair => counts.set(pair, (counts.get(pair) || 0) + 1)); }); const ranked = [...counts.entries()].sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0])); meta.textContent = `นำมารวมทุกแฟ้ม ${names.length} สูตร · พบคู่ที่ผ่านอย่างน้อย 1 สูตร ${ranked.length} คู่`; if (!ranked.length) { content.innerHTML = '

ยังไม่พบคู่ที่ผ่านเงื่อนไขจากสูตรที่บันทึกไว้

'; return; } content.innerHTML = ranked.map(([pair, count], index) => { const rankStyle = index === 0 ? 'bg-amber-100 border-amber-300 text-amber-950' : 'bg-white border-indigo-100 text-indigo-950'; return `
${index + 1}. ${pair}
พบ ${count} สูตร
`; }).join(''); }
